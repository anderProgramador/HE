# ============================================================================
# SAHE - T05 - Excepción por Puesto
#
# Formato con campos nombrados. Un registro por línea (también se acepta ¬
# como separador). Cada registro empieza con su tipo y sigue con pares
# clave=valor separados por |. Las líneas que empiezan con # son comentarios.
#
#   pantalla  titulo, subtitulo, tabla
#   grilla    columnas, porPagina, filtros, combos, pk, uk, rotulo, archivo,
#             acciones
#   ayuda     titulo, ancho, porPagina, columnas, destino  -> una por lupa
#   popup     titulo, ancho, deshabilitar
#   seccion   id, titulo, descripcion
#   campo     tipo, id, etiqueta, unidad, ayuda, ancho, lupa=<n ayuda>,
#             contador=si|<n>, y lo propio del tipo
#   accion    id, etiqueta, estilo, accion
#   mapeo     origen, campos
#
# Lo que devuelve dvpkg_spT05.cargar, en segmentos separados por ~ :
#   [0] combos   lista0 (lugares de trabajo) ¦ lista1 (estados)
#   [1] datos    filas de la grilla, en el orden del mapeo 'cargar'
#   [2] ayudas   catálogo de puestos, como lista plana |código|descripción|...
# ============================================================================

pantalla|tabla=T05|titulo=Excepción por Puesto|subtitulo=Puestos que no están sujetos al control de horas extra, según su lugar de trabajo.

# columnas:  titulo[:ancho[:tipo]] separadas por coma. Ancho 0 oculta la
#            columna, que es como se lleva la llave sin mostrarla.
#            tipo: texto | numero | decimal | fecha. Además de dar formato,
#            decide la alineación: texto a la izquierda, numero y decimal a la
#            derecha, fecha centrada.
# combos:    columna + tipo + lista. '3cs0' pone en la columna 3 un combo
#            simple con la lista 0; 'cm' sería de selección múltiple. Varios se
#            separan por coma. La columna con combo muestra la descripción.
# pk:        clave primaria, la que se manda a Obtener y a Eliminar.
# uk:        clave única. Se comprueba contra la grilla antes de grabar, para
#            avisar con los rótulos de la pantalla en vez de dejar que Oracle
#            devuelva el error del índice UNQ_T0501.
# rotulo:    campo que nombra a la fila en los mensajes, p. ej. al eliminar.
# acciones:  tope de lo que esta pantalla admite. Quién puede hacer cada una
#            llega en hdfTabla desde el menú, ya filtrada por el perfil.
grilla|columnas=Id:0:numero,Código:110,Puesto:340,Lugar de trabajo:220,Estado:100|porPagina=10|filtros=si|combos=3cs0,4cs1|pk=Id|uk=Puesto,LugarTrabajo|rotulo=DescripcionPuesto|archivo=ExcepcionPorPuesto.xlsx|acciones=nuevo,eliminarMultiple,excel,editar,eliminar,ver

# ---------------------------------------------------------------------------
# Ayuda 0: los 568 puestos del catálogo. Llegan con la carga, así que la lupa
# no vuelve a pedirle nada al servidor.
# destino: a qué campo va cada columna, por posición.
# ---------------------------------------------------------------------------
ayuda|titulo=Búsqueda de puestos|ancho=720|porPagina=10|columnas=Código:120,Puesto:440|destino=Puesto,DescripcionPuesto

# ---------------------------------------------------------------------------
# Ventana de mantenimiento
#
# deshabilitar: el puesto y el lugar identifican al registro; cambiarlos sería
# otro registro, no una modificación de este.
# ---------------------------------------------------------------------------
popup|titulo=Excepción por puesto|ancho=620|deshabilitar=editar:Puesto,DescripcionPuesto,LugarTrabajo

seccion|id=datos

# ---------------------------------------------------------------------------
# El orden de estos campos NO es cosmético: es el orden en que dvpkg_spT05
# .obtener devuelve el registro, y por lo tanto el orden en que se asignan los
# valores a los controles. Un oculto en medio de dos visibles conserva su
# lugar. Si se agrega, quita o mueve un campo aquí, hay que mover también su
# posición en la consulta.
#
#   1 accion          -> CodigoAccion      (oculto)
#   2 id              -> Id                (oculto)
#   3 cpuesto         -> Puesto
#   4 secuencia       -> Secuencia         (oculto)
#   5 dpuesto         -> DescripcionPuesto
#   6 dlugartrabajo   -> LugarTrabajo
#   7 estadofuncional -> Estado
#   8 creadopor       -> CreadoPor         (oculto)
#   9 fcreacion       -> FechaCreacion     (oculto)
#  10 actualizadopor  -> ActualizadoPor    (oculto)
# ---------------------------------------------------------------------------

campo|tipo=oculto|id=CodigoAccion
campo|tipo=oculto|id=Id|valor=0

# El código va en su propia fila: lleva la lupa y, en la fila siguiente, el
# nombre del puesto que devuelve la búsqueda.
campo|tipo=text|id=Puesto|etiqueta=Código del puesto|largo=60|ancho=medio|requerido=si|lupa=0|marcador=0039|ayuda=Búsquelo con la lupa; el nombre del puesto se completa solo.

campo|tipo=oculto|id=Secuencia|valor=0

campo|tipo=text|id=DescripcionPuesto|etiqueta=Puesto|largo=60|ancho=total|soloLectura=si|marcador=Se completa al elegir el código

campo|tipo=select|id=LugarTrabajo|etiqueta=Lugar de trabajo|ancho=medio|requerido=si|origen=cs|lista=0|ayuda=Dónde se aplica la excepción.
campo|tipo=select|id=Estado|etiqueta=Estado|ancho=medio|requerido=si|origen=cs|lista=1|ayuda=Inactivo deja de exceptuar al puesto.

campo|tipo=oculto|id=CreadoPor
campo|tipo=oculto|id=FechaCreacion
campo|tipo=oculto|id=ActualizadoPor

accion|id=Grabar|etiqueta=Grabar|estilo=primario|accion=grabar
accion|id=Cancelar|etiqueta=Cancelar|estilo=secundario|accion=cancelar

# Orden en que dvpkg_spT05.listar devuelve cada registro y en que la grilla
# muestra sus columnas.
mapeo|origen=cargar|campos=Id,Puesto,DescripcionPuesto,LugarTrabajo,Estado

# Orden que espera dvpkg_spT05.gestionar. La acción va primero porque el
# paquete la separa antes del primer |; después el controlador agrega el
# computerid y el loginid.
mapeo|origen=grabar|campos=CodigoAccion,Id,Puesto,Secuencia,LugarTrabajo,Estado,CreadoPor,ActualizadoPor
